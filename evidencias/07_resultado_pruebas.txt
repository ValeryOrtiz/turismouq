SQL> -- Se buscan habitaciones libres entre el 20/11/2026 y el 15/12/2026
SQL> DECLARE
  2      FUNCTION libre(p_hab NUMBER) RETURN BOOLEAN IS
  3          n NUMBER;
  4      BEGIN
  5          SELECT COUNT(*) INTO n
  6            FROM RESERVA_HABITACION rh JOIN RESERVA r ON r.reserva_id = rh.reserva_id
  7           WHERE rh.habitacion_id = p_hab AND r.estado <> 'CANCELADA'
  8             AND rh.fecha_checkin < DATE '2026-12-15' AND rh.fecha_checkout > DATE '2026-11-20';
  9          RETURN n = 0;
 10      END;
 11  BEGIN
 12      SELECT MIN(cliente_id) INTO :cli FROM CLIENTE;
 13      :hab_libre := NULL;
 14      :hab2      := NULL;
 15      :hab_otro  := NULL;
 16      FOR h IN (SELECT habitacion_id, capacidad_maxima, alojamiento_id
 17                  FROM HABITACION ORDER BY habitacion_id) LOOP
 18          IF :hab_libre IS NULL THEN
 19              IF h.capacidad_maxima >= 2 AND libre(h.habitacion_id) THEN
 20                  :hab_libre := h.habitacion_id; :cap := h.capacidad_maxima; :aloj_libre := h.alojamiento_id;
 21              END IF;
 22          ELSIF :hab2 IS NULL AND h.alojamiento_id = :aloj_libre THEN
 23              IF libre(h.habitacion_id) THEN :hab2 := h.habitacion_id; END IF;
 24          ELSIF :hab_otro IS NULL AND h.alojamiento_id <> :aloj_libre THEN
 25              IF libre(h.habitacion_id) THEN :hab_otro := h.habitacion_id; END IF;
 26          END IF;
 27          EXIT WHEN :hab_libre IS NOT NULL AND :hab2 IS NOT NULL AND :hab_otro IS NOT NULL;
 28      END LOOP;
 29      DBMS_OUTPUT.PUT_LINE('cliente=' || :cli || '  habitacion=' || :hab_libre || ' (capacidad ' || :cap ||
 30                           ', alojamiento ' || :aloj_libre || ')  hab2=' || :hab2 || '  hab_otro=' || :hab_otro);
 31  END;
 32  /
